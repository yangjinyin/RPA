# mysql执行过程

```c++
|-->mysql_alter_table()
|		|-->mysql_prepare_alter_table() //prepare ddl , 准备好列以及key定义
|		|-->fill_alter_inplace_info() //填充 handler_flags
|		|-->mysql_inplace_alter_table() || copy模式逻辑处理

|-->Sql_cmd_dml::execute()
|	|-->Sql_cmd_dml::execute_inner()
|	|	|-->Query_expression::optimize(): //优化器       
|	|	|	|-->Query_block::optimize()
|	|	|	|	|-->JOIN::optimize()
|	|	|	|	|	|-->JOIN::make_join_plan() //选择最佳访问路径    
|	|	|	|	|	|	|-->Optimize_table_order::choose_table_order()
|	|	|	|	|	|	|	|-->Optimize_table_order::greedy_search() //贪心搜索来寻得最小的连接代价    
|	|	|	|	|	|	|	|	|-->ptimize_table_order::best_extension_by_limited_search() //穷举搜索
|	|	|	|	|	|-->make_join_query_block() //对谓词条件（where连接条件，join中on条件）进行下推
|	|	|	|	|	|-->JOIN::create_access_paths() // 生成访问方式AccessPath，供后续迭代器Iterator访问使用
|	|	|	|	|	|	|-->JOIN::create_root_access_path_for_join()
|	|	|	|	|	|	|-->JOIN::attach_access_paths_for_having_and_limit()
|	|	|	|-->CreateIteratorFromAccessPath()  //根据path构建不同迭代器，生成表的不同访问方式
|	|	|-->accumulate_statement_cost(): //估算代价
|	|	|-->Query_expression::execute()  ||  explain_query()
|	|	|	|-->Query_expression::ExecuteIteratorQuery()//执行迭代器
|	|	|	|	|-->1. THD_STAGE_INFO()           //设置线程的状态为executing
|	|	|	|	|-->2. ClearForExecution()        //清理之前的迭代器数据
|	|	|	|	|-->3. query_result->start_execution(thd)        //设置为执行状态
|	|	|	|	|-->4. send_result_set_metadata()                //先把元数据发送给客户端(m_protocol...)
|	|	|	|	|-->5. m_root_iterator->Init()         //多态，调用不同的迭代器。此处我们拿hashjoin迭代器举例
|	|	|	|	|	|-->HashJoinIterator::Init()
|	|	|	|	|	|	|-->m_build_input->Init()  //内表初始化
|	|	|	|	|	|	|	|-->TableScanIterator::Init()
|	|	|	|	|	|	|	|	|-->handler::ha_rnd_init()
|	|	|	|	|	|	|-->HashJoinIterator::BuildHashTable() //读取内表数据进行建hashtable
|	|	|	|	|	|	|	|-->TableScanIterator::Read()
|	|	|	|	|	|	|	|	|-->handler::ha_rnd_next()
|	|	|	|	|	|	|	|	|	|-->ha_innobase::rnd_next() //读取buff放到m_row_buffer中
|	|	|	|	|	|	|-->HashJoinIterator::InitProbeIterator() //初始化外表探测
|	|	|	|	|	|	|	|-->TableScanIterator::Init()
|	|	|	|	|-->6. m_root_iterator->Read()
|	|	|	|	|	|-->HashJoinIterator::Read()
|	|	|	|	|	|	|-->HashJoinIterator::ReadRowFromProbeIterator()   //读取外表进行探测
|	|	|	|	|	|	|	|-->m_probe_input->Read()         //读取数据
|	|	|	|	|	|	|	|	|-->TableScanIterator::Read()
|	|	|	|	|	|	|	|-->HashJoinIterator::LookupProbeRowInHashTable() //在hashtable中寻找相同元素
|	|	|	|	|-->7. query_result->send_data() //每成功执行一条数据则发送给client
```
